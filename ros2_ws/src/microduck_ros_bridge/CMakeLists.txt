cmake_minimum_required(VERSION 3.16)
project(microduck_ros_bridge)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)

add_executable(microduck_bridge src/bridge_node.cpp)
target_compile_features(microduck_bridge PUBLIC cxx_std_17)
target_include_directories(microduck_bridge PRIVATE include)
set(ROS_DEPENDENCIES rclcpp geometry_msgs nav_msgs sensor_msgs diagnostic_msgs
  visualization_msgs tf2 tf2_ros tf2_geometry_msgs)
foreach(dependency IN LISTS ROS_DEPENDENCIES)
  list(APPEND ROS_INCLUDE_DIRS ${${dependency}_INCLUDE_DIRS})
  list(APPEND ROS_LIBRARIES ${${dependency}_LIBRARIES})
endforeach()
target_include_directories(microduck_bridge SYSTEM PRIVATE ${ROS_INCLUDE_DIRS})
target_link_libraries(microduck_bridge ${ROS_LIBRARIES} nlohmann_json::nlohmann_json)
install(TARGETS microduck_bridge DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_bridge test/test_bridge.cpp)
  target_include_directories(test_bridge PRIVATE include)
  target_link_libraries(test_bridge nlohmann_json::nlohmann_json)
endif()

ament_package()
